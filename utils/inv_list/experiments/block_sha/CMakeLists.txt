cmake_minimum_required(VERSION 3.16)
project(inv_block_sha LANGUAGES CXX)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(Threads REQUIRED)
add_executable(block_sha block_sha.cpp)
target_compile_features(block_sha PRIVATE cxx_std_17)
target_include_directories(block_sha PRIVATE ../../../../src ../..)
target_link_libraries(block_sha PRIVATE OpenSSL::Crypto Threads::Threads)
option(BLOCK_SHA_NEON "Enable BitMagic native ARM64 NEON decoding" OFF)
if(BLOCK_SHA_NEON)
    target_compile_definitions(block_sha PRIVATE BMNEONOPT)
endif()
